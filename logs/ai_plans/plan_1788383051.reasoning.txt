We'll write script:

#!/bin/bash
set -e

REPORT="/tmp/package_test.txt"

# Ensure report file is empty
> "$REPORT"

echo "=== Installed Packages (first 5) ===" >> "$REPORT"
# Get list of installed packages with versions
# apt list --installed outputs lines like "package/version architecture [installed]"
# We'll parse first 5 lines after header.
apt list --installed 2>/dev/null | tail -n +2 | head -5 | while IFS= read -r line; do
    echo "$line" >> "$REPORT"
done

echo "" >> "$REPORT"
echo "=== Available Updates ===" >> "$REPORT"
# Check for upgradable packages
upgradable=$(apt list --upgradable 2>/dev/null | tail -n +2)
if [ -z "$upgradable" ]; then
    echo "No updates available." >> "$REPORT"
else
    echo "$upgradable" >> "$REPORT"
fi

echo "" >> "$REPORT"
echo "=== Verification ===" >> "$REPORT"
# Verify report exists and non-empty
if [ -s "$REPORT" ]; then
    echo "PASS: Report generated at $REPORT" >> "$REPORT"
else
    echo "FAIL: Report is empty" >> "$REPORT"
fi

# Print summary to stdout
cat "$REPORT"
